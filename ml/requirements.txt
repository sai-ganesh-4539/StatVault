# Step 1: Navigate to your project directory
cd path/to/your/StatVault-ML

# Step 2: Initialize Git (if not already initialized)
git init

# Step 3: Set your Git identity
git config user.name "VasuML07"
git config user.email "your-email@example.com"

# Step 4: Add the remote repository
git remote add origin https://github.com/sai-ganesh-4539/StatVault.git

# Step 5: Install Git LFS (required for large model files)
# Download from: https://git-lfs.com/ if not installed
git lfs install

# Step 6: Track large file types
git lfs track "*.pkl"
git lfs track "*.onnx"
git lfs track "*.npy"
git lfs track "*.parquet"

# Step 7: Create .gitkeep files for empty directories you want to keep
mkdir -p data/raw data/features data/embeddings models
touch data/raw/.gitkeep
touch data/features/.gitkeep
touch data/embeddings/.gitkeep

# Step 8: Stage all files
git add .

# Step 9: Verify what will be committed
git status

# Step 10: Commit with message
git commit -m "feat: Add ML pipeline with models, features, and embeddings"

# Step 11: Pull latest changes (in case of conflicts)
git pull origin main --allow-unrelated-histories

# Step 12: Push to repository
git push -u origin main
